# ===========================================================================
# .htaccess für das Verzeichnis public/ auf IONOS Webspace
# ===========================================================================
# Diese Datei ersetzt public/.htaccess, wenn die Standardfassung von Laravel
# auf dem Webspace einen Serverfehler 500 auslöst.
#
# Hochladen als:  /intranet/public/.htaccess
# Der Dateiname beginnt mit einem Punkt. Falls Ihr Programm die Datei nicht
# anzeigt, im SFTP-Programm "versteckte Dateien anzeigen" aktivieren.
#
# Unterschiede zur Standardfassung von Laravel:
# 1. Die Anweisung "Options -MultiViews -Indexes" ist entfernt. Erlaubt der
#    Anbieter das Setzen von Options im Verzeichnis nicht, antwortet Apache mit
#    500, und zwar bei JEDEM Aufruf. Das ist die häufigste Ursache auf
#    Webspace-Paketen. Der Verzeichnisschutz wird stattdessen über
#    DirectoryIndex und die Umschreiberegeln erreicht.
# 2. RewriteBase ist gesetzt. Zeigt die Domain auf ein Unterverzeichnis, finden
#    die Regeln ohne diese Angabe den Front Controller nicht zuverlässig.
# 3. Die Weiterleitung von HTTP auf HTTPS ist enthalten, weil die Anmeldung
#    HTTPS voraussetzt (das Sitzungs-Cookie ist als "secure" gesetzt).
# ===========================================================================

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /

    # HTTP auf HTTPS umleiten. Bei IONOS steht das Protokoll in HTTPS bzw. in
    # der Weiterleitungs-Kopfzeile des vorgeschalteten Servers.
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !https
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # Kopfzeile Authorization durchreichen
    RewriteCond %{HTTP:Authorization} .
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # Kopfzeile X-XSRF-Token durchreichen
    RewriteCond %{HTTP:x-xsrf-token} .
    RewriteRule .* - [E=HTTP_X_XSRF_TOKEN:%{HTTP:X-XSRF-Token}]

    # Abschließenden Schrägstrich entfernen, sofern kein Verzeichnis
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_URI} (.+)/$
    RewriteRule ^ %1 [L,R=301]

    # Alle übrigen Anfragen an den Front Controller
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteRule ^ index.php [L]
</IfModule>

DirectoryIndex index.php

# Verzeichnisauflistung unterbinden, ohne die Anweisung Options zu verwenden
<IfModule mod_autoindex.c>
    IndexIgnore *
</IfModule>

# Zugriff auf Dateien verhindern, die nie ausgeliefert werden dürfen.
# Sie liegen ohnehin außerhalb von public, das ist eine zusätzliche Absicherung.
<FilesMatch "^\.(env|git|htaccess)|composer\.(json|lock)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>
